using Microsoft.AspNetCore.Mvc; using ShoppingCartApp.Models; using System.Text.Json;
namespace ShoppingCartApp.Controllers
{
public class ShoppingController : Controller
{
private static List<Product> products = new List<Product>
{
new Product { Id = 1, Name = "Laptop", Price = 55000, ImageUrl="https://via.placeholder.com/150"
},
new Product { Id = 2, Name = "Mouse", Price = 600, ImageUrl="https://via.placeholder.com/150"
},
new Product { Id = 3, Name = "Keyboard", Price = 1200, ImageUrl="https://via.placeholder.com/150"
},
new Product { Id = 4, Name = "Monitor", Price = 9000, ImageUrl="https://via.placeholder.com/150"
}
};

public IActionResult Index() => View(products);
public IActionResult Cart() => View(GetCart());
public IActionResult AddToCart(int id)
{
var cart = GetCart(); var product =
products.FirstOrDefault(p => p.Id == id);
if (product != null)
{
var item = cart.FirstOrDefault(c => c.Product.Id == id);
if (item == null) cart.Add(new CartItem { Product = product, Quantity = 1 });
else item.Quantity++;
}
SaveCart(cart); return
RedirectToAction("Index");
}
public IActionResult Remove(int id)
{
var cart = GetCart(); var item =
cart.FirstOrDefault(c => c.Product.Id == id);
if (item != null) cart.Remove(item);
SaveCart(cart); return
RedirectToAction("Cart");
}
public IActionResult ClearCart()
{
HttpContext.Session.Remove("Cart");
return RedirectToAction("Cart");
}
private List<CartItem> GetCart()
{
var cart = HttpContext.Session.GetString("Cart");
if (string.IsNullOrEmpty(cart)) return new List<CartItem>();
return JsonSerializer.Deserialize<List<CartItem>> (cart)!;
}
private void SaveCart(List<CartItem> cart)
{
HttpContext.Session.SetString("Cart", JsonSerializer.Serialize(cart));
}
}
}

public class MovieController : Controller

cache) => _cache = cache;

public IActionResult SelectTheatre(string theatre)
{
HttpContext.Session.SetString("Theatre", theatre);
return RedirectToAction("Showtimes");
}
public IActionResult Showtimes()
{
string theatre = HttpContext.Session.GetString("Theatre") ??
"";
string message;
if (!_cache.TryGetValue(theatre, out object? data))
{
var timings = new { Timings = new[] { "9:00 AM", "12:00 PM", "5:00 PM" },
Movies =
new[] { "ABC", "PQR", "XYZ" } };
_cache.Set(theatre, timings, TimeSpan.FromSeconds(30));
message = "Loaded fresh (not from cache)";
ViewBag.Data = timings;
}
else
{
message = "Reading From Cache"; ViewBag.Data = data;
}
ViewBag.Message = message; return View();
}
}

NotFound();

student.Id = students.Count +
1;
students.Add(student); return
CreatedAtAction(nameof(GetStudent), new {
id = student.Id }, student);
}

UpdateStudent(int id, Student updatedStudent)

student.Course = updatedStudent.Course;
student.Age = updatedStudent.Age;
return Ok(student);
}
