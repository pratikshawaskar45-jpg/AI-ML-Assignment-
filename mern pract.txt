#Practical 3 Text to music generation
!pip install transformers scipy
from transformers import AutoProcessor, MusicgenForConditionalGeneration
import scipy.io.wavfile

processor = AutoProcessor.from_pretrained("facebook/musicgen-small")
model = MusicgenForConditionalGeneration.from_pretrained("facebook/musicgen-small")

prompt=input("Enatr a music description: ")

inputs=processor(text=[prompt], return_tensors="pt")
audio = model.generate(
    **inputs,
    max_new_tokens=265
)
scipy.io.wavfile.write("music.wav", 32000,audio[0,0].numpy())
print("Music generated and save as music.wav")


#Practical 4 (Image to text generation)
!pip install transformers pillow
from transformers import BlipProcessor, BlipForConditionalGeneration
from PIL import Image

processor = BlipProcessor.from_pretrained("Salesforce/blip-image-captioning-large")
model =BlipForConditionalGeneration.from_pretrained("Salesforce/blip-image-captioning-large")

image_path=input("Enter image Path: ")
image=Image.open(image_path).convert("RGB")
inputs = processor(images=image, return_tensors="pt")
output = model.generate(
    **inputs,
    max_new_tokens=50
)
print("DEscription ", processor.decode(output[0], skip_special_tokens=True))
